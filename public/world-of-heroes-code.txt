/**

import { useState, useEffect } from 'react';
import { WelcomeScreen } from './components/WelcomeScreen';
import { ParentAuthScreen } from './components/ParentAuthScreen';
import { HeroesScreen } from './components/HeroesScreen';
import { HomeScreen } from './components/HomeScreen';
import { ChallengesScreen } from './components/ChallengesScreen';
import { RewardsScreen } from './components/RewardsScreen';
import { StarCelebrationModal } from './components/StarCelebrationModal';
import { TopMotivationalToast } from './components/TopMotivationalToast';
import { ParentRewardRequestsModal } from './components/ParentRewardRequestsModal';
import { getRandomMotivationalMessage } from './data/motivationalMessages';
import { DAILY_TASKS, HEROES, REWARDS } from './data/appData';
import { Task, Reward, ParentChildProfile, CustomRewardRequest } from './types';
import { playPopSound } from './utils/audio';

type Screen = 'welcome' | 'parent_auth' | 'heroes' | 'home' | 'challenges' | 'rewards';

const STORAGE_KEYS = {
  STARS: 'world_of_heroes_stars',
  HERO: 'world_of_heroes_selected_hero',
  COMPLETED: 'world_of_heroes_completed_tasks',
  LAST_DATE: 'world_of_heroes_last_date',
  PROFILE: 'world_of_heroes_parent_child_profile',
  CUSTOM_REQUESTS: 'world_of_heroes_custom_reward_requests',
  REWARDS: 'world_of_heroes_rewards_list',
};

export default function App() {
  const [currentScreen, setCurrentScreen] = useState<Screen>('welcome');
  const [returnScreenAfterAuth, setReturnScreenAfterAuth] = useState<Screen>('heroes');
  const [selectedHeroId, setSelectedHeroId] = useState<string>('hero-blue');
  const [stars, setStars] = useState<number>(0);
  const [completedTaskIds, setCompletedTaskIds] = useState<string[]>([]);
  const [celebrationOpen, setCelebrationOpen] = useState<boolean>(false);
  const [celebratingTask, setCelebratingTask] = useState<Task | null>(null);
  const [showParentModal, setShowParentModal] = useState<boolean>(false);
  const [showParentRequestsModal, setShowParentRequestsModal] = useState<boolean>(false);

  // Mother Login & Child Name Profile
  const [profile, setProfile] = useState<ParentChildProfile | null>(null);

  // Rewards list
  const [rewardsList, setRewardsList] = useState<Reward[]>(REWARDS);

  // Child's custom gift requests
  const [customRequests, setCustomRequests] = useState<CustomRewardRequest[]>([]);

  // Top motivational message when completing any task
  const [motivationalMessage, setMotivationalMessage] = useState<string | null>(null);
  const [motivationalTaskTitle, setMotivationalTaskTitle] = useState<string | undefined>(undefined);

  // Initialize and load saved state from localStorage
  useEffect(() => {
    try {
      const savedStars = localStorage.getItem(STORAGE_KEYS.STARS);
      if (savedStars !== null) {
        setStars(parseInt(savedStars, 10) || 0);
      }

      const savedHero = localStorage.getItem(STORAGE_KEYS.HERO);
      if (savedHero && HEROES.some((h) => h.id === savedHero)) {
        setSelectedHeroId(savedHero);
      }

      const savedProfile = localStorage.getItem(STORAGE_KEYS.PROFILE);
      if (savedProfile) {
        setProfile(JSON.parse(savedProfile));
      }

      const savedRewards = localStorage.getItem(STORAGE_KEYS.REWARDS);
      if (savedRewards) {
        const parsed: Reward[] = JSON.parse(savedRewards);
        // Ensure removed rewards (chocolate & favorite meal) do not persist from older localStorage
        const filtered = parsed.filter((r) => r.id !== 'r3' && r.id !== 'r4');
        setRewardsList(filtered);
        localStorage.setItem(STORAGE_KEYS.REWARDS, JSON.stringify(filtered));
      }

      const savedCustomReqs = localStorage.getItem(STORAGE_KEYS.CUSTOM_REQUESTS);
      if (savedCustomReqs) {
        setCustomRequests(JSON.parse(savedCustomReqs));
      }

      const todayStr = new Date().toISOString().split('T')[0];
      const savedDate = localStorage.getItem(STORAGE_KEYS.LAST_DATE);

      if (savedDate === todayStr) {
        const savedCompleted = localStorage.getItem(STORAGE_KEYS.COMPLETED);
        if (savedCompleted) {
          setCompletedTaskIds(JSON.parse(savedCompleted));
        }
      } else {
        localStorage.setItem(STORAGE_KEYS.LAST_DATE, todayStr);
        localStorage.setItem(STORAGE_KEYS.COMPLETED, JSON.stringify([]));
        setCompletedTaskIds([]);
      }
    } catch (e) {
      console.error('Failed to read localStorage:', e);
    }
  }, []);

  const updateStars = (newStars: number) => {
    setStars(newStars);
    try {
      localStorage.setItem(STORAGE_KEYS.STARS, newStars.toString());
    } catch (e) {
      console.error(e);
    }
  };

  const handleAdjustStars = (delta: number) => {
    const nextStars = Math.max(0, stars + delta);
    updateStars(nextStars);
  };

  const handleResetDailyTasks = () => {
    setCompletedTaskIds([]);
    try {
      localStorage.setItem(STORAGE_KEYS.COMPLETED, JSON.stringify([]));
    } catch (e) {
      console.error(e);
    }
  };

  const handleSaveProfile = (newProfile: ParentChildProfile) => {
    setProfile(newProfile);
    try {
      localStorage.setItem(STORAGE_KEYS.PROFILE, JSON.stringify(newProfile));
    } catch (e) {
      console.error(e);
    }
    setCurrentScreen(returnScreenAfterAuth);
  };

  const handleHeroSelect = (heroId: string) => {
    setSelectedHeroId(heroId);
    try {
      localStorage.setItem(STORAGE_KEYS.HERO, heroId);
    } catch (e) {
      console.error(e);
    }
  };

  const handleCompleteTask = (task: Task) => {
    const isAlreadyCompleted = completedTaskIds.includes(task.id);

    if (!isAlreadyCompleted) {
      const nextStars = stars + 1;
      updateStars(nextStars);

      const nextCompleted = [...completedTaskIds, task.id];
      setCompletedTaskIds(nextCompleted);
      try {
        localStorage.setItem(STORAGE_KEYS.COMPLETED, JSON.stringify(nextCompleted));
      } catch (e) {
        console.error(e);
      }
    }

    setCelebratingTask(task);
    setCelebrationOpen(true);
  };

  const handleCloseCelebration = () => {
    setCelebrationOpen(false);
    // Trigger motivational message at the top right after the child takes the star!
    const randomMsg = getRandomMotivationalMessage(profile?.childFirstName);
    setMotivationalMessage(randomMsg);
    setMotivationalTaskTitle(celebratingTask?.title);
  };

  const handleRedeemReward = (reward: Reward) => {
    if (stars >= reward.cost) {
      const nextStars = Math.max(0, stars - reward.cost);
      updateStars(nextStars);

      setRewardsList((prev) => {
        const next = prev.map((r) =>
          r.id === reward.id ? { ...r, redeemedCount: (r.redeemedCount || 0) + 1 } : r
        );
        try {
          localStorage.setItem(STORAGE_KEYS.REWARDS, JSON.stringify(next));
        } catch (e) {
          console.error(e);
        }
        return next;
      });
    }
  };

  // Custom Gift Request Handlers
  const handleCreateCustomRequest = (
    req: Omit<CustomRewardRequest, 'id' | 'createdAt' | 'status'>
  ) => {
    const newRequest: CustomRewardRequest = {
      ...req,
      id: `req-${Date.now()}`,
      status: 'pending',
      createdAt: new Date().toISOString(),
    };
    const nextRequests = [newRequest, ...customRequests];
    setCustomRequests(nextRequests);
    try {
      localStorage.setItem(STORAGE_KEYS.CUSTOM_REQUESTS, JSON.stringify(nextRequests));
    } catch (e) {
      console.error(e);
    }
  };

  const handleApproveCustomRequest = (requestId: string, approvedCost: number) => {
    const targetReq = customRequests.find((r) => r.id === requestId);
    if (!targetReq) return;

    // 1. Update request status to approved
    const nextRequests = customRequests.map((r) =>
      r.id === requestId
        ? { ...r, status: 'approved' as const, approvedCost, reviewedAt: new Date().toISOString() }
        : r
    );
    setCustomRequests(nextRequests);
    try {
      localStorage.setItem(STORAGE_KEYS.CUSTOM_REQUESTS, JSON.stringify(nextRequests));
    } catch (e) {
      console.error(e);
    }

    // 2. Add or update as a custom Reward in the rewardsList
    const customRewardId = `custom-reward-${requestId}`;
    setRewardsList((prev) => {
      const existingIndex = prev.findIndex(
        (r) => r.customRequestId === requestId || r.id === customRewardId
      );
      let nextRewards: Reward[];
      if (existingIndex >= 0) {
        nextRewards = prev.map((r, i) =>
          i === existingIndex
            ? { ...r, cost: approvedCost, title: targetReq.title, icon: targetReq.icon }
            : r
        );
      } else {
        const newReward: Reward = {
          id: customRewardId,
          title: targetReq.title,
          icon: targetReq.icon,
          cost: approvedCost,
          redeemedCount: 0,
          isCustom: true,
          customRequestId: requestId,
        };
        // Add at the beginning of the list for high visibility
        nextRewards = [newReward, ...prev];
      }

      try {
        localStorage.setItem(STORAGE_KEYS.REWARDS, JSON.stringify(nextRewards));
      } catch (e) {
        console.error(e);
      }
      return nextRewards;
    });
  };

  const handleRejectCustomRequest = (requestId: string) => {
    const nextRequests = customRequests.map((r) =>
      r.id === requestId
        ? { ...r, status: 'rejected' as const, reviewedAt: new Date().toISOString() }
        : r
    );
    setCustomRequests(nextRequests);
    try {
      localStorage.setItem(STORAGE_KEYS.CUSTOM_REQUESTS, JSON.stringify(nextRequests));
    } catch (e) {
      console.error(e);
    }
  };

  const handleDeleteCustomRequest = (requestId: string) => {
    const nextRequests = customRequests.filter((r) => r.id !== requestId);
    setCustomRequests(nextRequests);
    try {
      localStorage.setItem(STORAGE_KEYS.CUSTOM_REQUESTS, JSON.stringify(nextRequests));
    } catch (e) {
      console.error(e);
    }

    // Also remove from rewards if it was approved
    setRewardsList((prev) => {
      const nextRewards = prev.filter(
        (r) => r.customRequestId !== requestId && r.id !== `custom-reward-${requestId}`
      );
      try {
        localStorage.setItem(STORAGE_KEYS.REWARDS, JSON.stringify(nextRewards));
      } catch (e) {
        console.error(e);
      }
      return nextRewards;
    });
  };

  const pendingCustomRequestsCount = customRequests.filter((r) => r.status === 'pending').length;

  return (
    <div className="min-h-screen w-full bg-[#D8C6B7] text-[#2B1B15] font-sans antialiased" dir="rtl">
      {currentScreen === 'welcome' && (
        <WelcomeScreen
          profile={profile}
          onOpenAuth={() => {
            setReturnScreenAfterAuth('heroes');
            setCurrentScreen('parent_auth');
          }}
          onStart={() => {
            if (!profile) {
              setReturnScreenAfterAuth('heroes');
              setCurrentScreen('parent_auth');
            } else {
              setCurrentScreen('heroes');
            }
          }}
        />
      )}

      {currentScreen === 'parent_auth' && (
        <ParentAuthScreen
          initialProfile={profile}
          onSaveProfile={handleSaveProfile}
          onBack={profile ? () => setCurrentScreen(returnScreenAfterAuth) : () => setCurrentScreen('welcome')}
        />
      )}

      {currentScreen === 'heroes' && (
        <HeroesScreen
          selectedHeroId={selectedHeroId}
          onSelectHero={handleHeroSelect}
          onConfirm={() => setCurrentScreen('home')}
        />
      )}

      {currentScreen === 'home' && (
        <HomeScreen
          stars={stars}
          selectedHeroId={selectedHeroId}
          tasks={DAILY_TASKS}
          completedTaskIds={completedTaskIds}
          profile={profile}
          onCompleteTask={handleCompleteTask}
          onOpenExtraChallenges={() => setCurrentScreen('challenges')}
          onOpenRewards={() => setCurrentScreen('rewards')}
          onChangeHero={() => setCurrentScreen('heroes')}
          onOpenParentAuth={() => {
            setReturnScreenAfterAuth('home');
            setCurrentScreen('parent_auth');
          }}
          onAdjustStars={handleAdjustStars}
          onResetDailyTasks={handleResetDailyTasks}
        />
      )}

      {currentScreen === 'challenges' && (
        <ChallengesScreen
          stars={stars}
          completedTaskIds={completedTaskIds}
          onCompleteTask={handleCompleteTask}
          onBack={() => setCurrentScreen('home')}
        />
      )}

      {currentScreen === 'rewards' && (
        <RewardsScreen
          stars={stars}
          rewards={rewardsList}
          customRequests={customRequests}
          profile={profile}
          onRedeemReward={handleRedeemReward}
          onEditProfile={() => {
            setReturnScreenAfterAuth('rewards');
            setCurrentScreen('parent_auth');
          }}
          onBack={() => setCurrentScreen('home')}
          onCreateCustomRequest={handleCreateCustomRequest}
          onApproveCustomRequest={handleApproveCustomRequest}
          onRejectCustomRequest={handleRejectCustomRequest}
          onDeleteCustomRequest={handleDeleteCustomRequest}
        />
      )}

      {/* Top Random Motivational Message Toast when child receives star and returns to tasks */}
      <TopMotivationalToast
        message={motivationalMessage}
        taskTitle={motivationalTaskTitle}
        onClose={() => setMotivationalMessage(null)}
      />

      <StarCelebrationModal
        isOpen={celebrationOpen}
        onClose={handleCloseCelebration}
        taskTitle={celebratingTask?.title}
        totalStars={stars}
      />

      {/* Discreet floating button for parents: زر صغير خالص للأهل */}
      {(currentScreen === 'home' || currentScreen === 'challenges') && (
        <button
          onClick={() => {
            playPopSound();
            setShowParentModal(true);
          }}
          title="خيارات ولي الأمر"
          className="fixed bottom-3 right-3 z-40 w-8 h-8 rounded-full bg-black/40 hover:bg-black/60 text-white flex items-center justify-center text-xs opacity-40 hover:opacity-100 transition-opacity cursor-pointer shadow-md"
        >
          ⚙️
          {pendingCustomRequestsCount > 0 && (
            <span className="absolute -top-1 -right-1 w-3.5 h-3.5 bg-red-600 rounded-full border-2 border-white animate-pulse" />
          )}
        </button>
      )}

      {/* Parent Quick Controls Modal */}
      {showParentModal && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-xs">
          <div className="w-full max-w-xs rounded-3xl p-5 bg-[#FAF5EF] border-4 border-black text-center text-[#2B1B15] shadow-2xl animate-in zoom-in-95 duration-150 space-y-3">
            <h4 className="text-xl font-black mb-1">⚙️ خيارات ولي الأمر</h4>

// Bulletproof Audio Engine with Real Sound Synthesis, Direct Web Audio, and Resilient Voice Reader

let audioCtx: AudioContext | null = null;
let isMuted: boolean = false;

export function getIsMuted(): boolean {
  return isMuted;
}

export function toggleMute(): boolean {
  isMuted = !isMuted;
  if (isMuted) {
    stopSpeaking();
  }
  return isMuted;
}

/**
 * Initializes and unlocks AudioContext reliably on user gestures
 */
export function ensureAudioUnlocked(): AudioContext | null {
  if (typeof window === 'undefined') return null;

  try {
    const AudioCtxClass =
      window.AudioContext ||
      (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;

    if (!AudioCtxClass) return null;

    if (!audioCtx) {
      audioCtx = new AudioCtxClass();
    }

    if (audioCtx.state === 'suspended') {
      audioCtx.resume();
    }

    return audioCtx;
  } catch (err) {
    console.error('AudioContext init error:', err);
    return null;
  }
}

// Global click/touch listener to auto-unlock audio on any first interaction
if (typeof window !== 'undefined') {
  const unlockEvents = ['click', 'touchstart', 'touchend', 'keydown'];
  const unlocker = () => {
    ensureAudioUnlocked();
    if (window.speechSynthesis && window.speechSynthesis.paused) {
      window.speechSynthesis.resume();
    }
  };
  unlockEvents.forEach((evt) => {
    window.addEventListener(evt, unlocker, { passive: true, once: false });
  });
}

/**
 * 1. Play Cartoon Spring Boing
 */
export function playBoingSound() {
  if (isMuted) return;
  const ctx = ensureAudioUnlocked();
  if (!ctx) return;

  try {
    const now = ctx.currentTime;
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();

    osc.type = 'triangle';
    osc.frequency.setValueAtTime(160, now);
    osc.frequency.exponentialRampToValueAtTime(540, now + 0.15);
    osc.frequency.exponentialRampToValueAtTime(320, now + 0.32);

    gain.gain.setValueAtTime(0.4, now);
    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.35);

    osc.connect(gain);
    gain.connect(ctx.destination);

    osc.start(now);
    osc.stop(now + 0.36);
  } catch (e) {
    console.error(e);
  }
}

/**
 * Distinct, loud Cartoon Character Syllable Babble
 * Guaranteed to produce clear, expressive sound in every browser via Web Audio API!
 */
export function playCartoonBabble(
  character: 'bondok' | 'basbousa' | 'ammi_felfel' | 'robot',
  syllablesCount: number = 6,
  onEnd?: () => void
) {
  if (isMuted) {
    if (onEnd) onEnd();
    return;
  }

  const ctx = ensureAudioUnlocked();
  if (!ctx) {
    if (onEnd) onEnd();
    return;
  }

  try {
    const now = ctx.currentTime;
    let basePitch = 480;
    let waveType: OscillatorType = 'triangle';

    if (character === 'bondok') {
      basePitch = 500; // energetic rabbit
      waveType = 'triangle';
    } else if (character === 'basbousa') {
      basePitch = 600; // cute smart cat
      waveType = 'sine';
    } else if (character === 'ammi_felfel') {
      basePitch = 240; // jovial shopkeeper
      waveType = 'sawtooth';
    } else if (character === 'robot') {
      basePitch = 340;
      waveType = 'square';
    }

    const count = Math.min(Math.max(syllablesCount, 4), 10);
    const duration = 0.11;

    for (let i = 0; i < count; i++) {
      const t = now + i * (duration * 0.96);
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();

      const pitchOffset = Math.sin((i / count) * Math.PI) * 140 + (Math.random() * 60 - 30);
      const freq = basePitch + pitchOffset;

      osc.type = waveType;
      osc.frequency.setValueAtTime(freq, t);
      osc.frequency.exponentialRampToValueAtTime(freq * 1.2, t + duration * 0.4);
      osc.frequency.exponentialRampToValueAtTime(freq * 0.95, t + duration);

      gain.gain.setValueAtTime(0.001, t);
      gain.gain.linearRampToValueAtTime(0.28, t + 0.015);
      gain.gain.exponentialRampToValueAtTime(0.001, t + duration);

      osc.connect(gain);
      gain.connect(ctx.destination);

      osc.start(t);
      osc.stop(t + duration + 0.02);
    }

    const totalMs = count * duration * 1000;
    setTimeout(() => {
      if (onEnd) onEnd();
    }, totalMs + 40);
  } catch (err) {
    console.error(err);
    if (onEnd) onEnd();
  }
}

/**
 * Universal Cartoon Voice Player
 */
export function speakCartoonDialogue(
  text: string,
  character: 'bondok' | 'basbousa' | 'ammi_felfel' | 'robot' = 'bondok',
  onEnd?: () => void
) {
  if (isMuted) {
    if (onEnd) onEnd();
    return;
  }

  ensureAudioUnlocked();

  if (activeVoiceMode === 'cartoon_sfx') {
    const wordCount = text.trim().split(/\s+/).length;
    const syllables = Math.min(Math.max(wordCount, 4), 10);
    playCartoonBabble(character, syllables, onEnd);
    return;
  }

  // If in 'natural_human' mode:
  if (typeof window === 'undefined' || !window.speechSynthesis) {
    playCartoonBabble(character, 6, onEnd);
    return;
  }

  try {
    window.speechSynthesis.cancel();

    const utterance = new SpeechSynthesisUtterance(text);
    const voices = window.speechSynthesis.getVoices();
    const arVoice = voices.find(
      (v) => v.lang.startsWith('ar') || v.lang.includes('EG') || v.lang.includes('SA')
    );

    if (arVoice) {
      utterance.voice = arVoice;
    }
    utterance.lang = 'ar-EG';

    if (character === 'bondok') {
      utterance.pitch = 1.35;
      utterance.rate = 1.1;
    } else if (character === 'basbousa') {
      utterance.pitch = 1.25;
      utterance.rate = 1.0;
    } else if (character === 'ammi_felfel') {
      utterance.pitch = 0.95;
      utterance.rate = 0.95;
    }

    utterance.onend = () => {
      if (onEnd) onEnd();
    };

    utterance.onerror = () => {
      // Fallback to cartoon babble if TTS fails or is blocked
      playCartoonBabble(character, 5, onEnd);
    };

    window.speechSynthesis.speak(utterance);
  } catch (err) {
    console.error(err);
    playCartoonBabble(character, 5, onEnd);
  }
}

export function stopSpeaking() {
  if (typeof window !== 'undefined' && window.speechSynthesis) {
    window.speechSynthesis.cancel();
  }
}

import { Task, Hero, ParentChildProfile } from '../types';

interface HomeScreenProps {
  stars: number;
  selectedHeroId: string;
  tasks: Task[];
  completedTaskIds: string[];
  profile?: ParentChildProfile | null;
  onCompleteTask: (task: Task) => void;
  onOpenExtraChallenges: () => void;
  onOpenRewards: () => void;
  onChangeHero: () => void;
  onOpenParentAuth?: () => void;
  onAdjustStars?: (delta: number) => void;
  onResetDailyTasks?: () => void;
}

export const HomeScreen: React.FC<HomeScreenProps> = ({
  stars,
  selectedHeroId,
  tasks,
  completedTaskIds,
  profile,
  onCompleteTask,
  onOpenExtraChallenges,
  onOpenRewards,
  onChangeHero,
  onOpenParentAuth,
  onAdjustStars,
  onResetDailyTasks,
}) => {
  const currentHero: Hero = HEROES.find((h) => h.id === selectedHeroId) || HEROES[0];
  const [animatingTaskId, setAnimatingTaskId] = React.useState<string | null>(null);
  const [showParentControls, setShowParentControls] = React.useState<boolean>(false);

  // Animated star counter state for interactive escalation
  const [displayedStars, setDisplayedStars] = React.useState<number>(stars);
  const [isIncrementing, setIsIncrementing] = React.useState<boolean>(false);
  const [starGainDelta, setStarGainDelta] = React.useState<number | null>(null);
  const [showSparkles, setShowSparkles] = React.useState<boolean>(false);
  const prevStarsRef = React.useRef<number>(stars);

  React.useEffect(() => {
    if (stars > prevStarsRef.current) {
      const delta = stars - prevStarsRef.current;
      setStarGainDelta(delta);
      setIsIncrementing(true);
      setShowSparkles(true);

      // Stepping counter up smoothly so the child sees the numbers climbing
      const startVal = prevStarsRef.current;
      const targetVal = stars;
      const stepDuration = Math.min(80, Math.max(30, 400 / delta));
      let currentVal = startVal;

      const interval = setInterval(() => {
        currentVal += 1;
        setDisplayedStars(currentVal);
        if (currentVal >= targetVal) {
          clearInterval(interval);
        }
      }, stepDuration);

      const resetTimer = setTimeout(() => {
        setIsIncrementing(false);
        setStarGainDelta(null);
        setShowSparkles(false);
      }, 1300);

      prevStarsRef.current = stars;
      return () => {
        clearInterval(interval);
        clearTimeout(resetTimer);
      };
    } else {
      setDisplayedStars(stars);
      prevStarsRef.current = stars;
    }
  }, [stars]);

  const childDisplayName = profile
    ? `${profile.childFirstName}${profile.childSecondName ? ` ${profile.childSecondName}` : ''}`
    : '';

  const handleTaskClick = (task: Task) => {
    setAnimatingTaskId(task.id);
    setTimeout(() => {
      setAnimatingTaskId((prev) => (prev === task.id ? null : prev));
    }, 700);
    onCompleteTask(task);
  };

  return (
    <div className="relative min-h-screen w-full flex flex-col justify-between items-center py-6 px-4 bg-[#D8C6B7] text-[#2B1B15] select-none">
      
      {/* Top Header matching home.png */}
      <div className="w-full max-w-sm flex items-center justify-between gap-2 pt-1 pb-3">
        {/* Star Counter Pill on Left with Interactive Escalation */}
        <div
          className={`relative flex items-center gap-1.5 bg-white/95 border-[2.5px] rounded-full px-3.5 py-1.5 shadow-sm transition-all duration-300 ${
            isIncrementing ? 'animate-star-pill-pulse scale-105 border-amber-600' : 'border-black'
          }`}
        >
          {/* Floating +N Stars badge animation */}
          {starGainDelta !== null && starGainDelta > 0 && (
            <div
              key={`gain-${displayedStars}`}
              className="absolute -top-7 left-1/2 -translate-x-1/2 flex items-center gap-1 bg-amber-400 text-amber-950 font-black text-xs px-2.5 py-0.5 rounded-full border-2 border-black comic-shadow-sm pointer-events-none z-30 animate-gain-fly-up whitespace-nowrap shadow-md"
            >
              <span>+{starGainDelta}</span>
              <span className="text-sm">⭐</span>
            </div>
          )}

          {/* Sparkles around pill */}
          {showSparkles && (
            <div className="absolute inset-0 pointer-events-none z-20">
              <span className="absolute -top-2 -right-1 text-sm animate-sparkle-burst-1">✨</span>
              <span className="absolute -bottom-2 -left-1 text-xs animate-sparkle-burst-2">🌟</span>
              <span className="absolute -top-1 -left-2 text-xs animate-sparkle-burst-3">⭐</span>
            </div>
          )}

          {/* Animated Ascending Number */}
          <span
            key={displayedStars}
            className={`text-2xl font-black text-[#2B1B15] tracking-wider min-w-[28px] text-center inline-block ${
              isIncrementing ? 'animate-star-num-pop' : ''
            }`}
          >
            {displayedStars}
          </span>

          {/* Glowing smiling star icon */}
          <div
            className={`relative w-7 h-7 flex items-center justify-center ${
              isIncrementing ? 'animate-star-celebrate' : 'animate-star-bounce'
            }`}
          >
            <svg width="28" height="28" viewBox="0 0 100 100">
              <polygon
                points="50,5 64,34 96,39 73,62 78,94 50,79 22,94 27,62 4,39 36,34"
                fill="#FFD200"
                stroke="#E5A100"
                strokeWidth="4"
              />
              <circle cx="43" cy="46" r="3.5" fill="#2B1810" />
              <circle cx="57" cy="46" r="3.5" fill="#2B1810" />
              <path d="M 45 54 Q 50 60 55 54" fill="none" stroke="#2B1810" strokeWidth="3" strokeLinecap="round" />
            </svg>
          </div>

          {/* Tiny Parent Quick Controls Button */}
          <button
            onClick={() => {
              playPopSound();
              setShowParentControls(true);
            }}
            title="تحكم الأهل: تعديل النجوم وإعادة مهام اليوم"
            className="w-5 h-5 rounded-full bg-amber-200 hover:bg-amber-300 border border-black/70 flex items-center justify-center text-[10px] cursor-pointer transition-transform active:scale-90 mr-0.5 shadow-2xs"

import React, { useState } from 'react';
import { CustomRewardRequest } from '../types';
import { playPopSound, playFanfareSound } from '../utils/audio';

interface ParentRewardRequestsModalProps {
  isOpen: boolean;
  requests: CustomRewardRequest[];
  onClose: () => void;
  onApproveRequest: (requestId: string, approvedCost: number) => void;
  onRejectRequest: (requestId: string) => void;
  onDeleteRequest: (requestId: string) => void;
}

export const ParentRewardRequestsModal: React.FC<ParentRewardRequestsModalProps> = ({
  isOpen,
  requests,
  onClose,
  onApproveRequest,
  onRejectRequest,
  onDeleteRequest,
}) => {
  // Store customized cost per request before approving
  const [editedCosts, setEditedCosts] = useState<Record<string, number>>({});
  const [activeTab, setActiveTab] = useState<'pending' | 'all'>('pending');

  if (!isOpen) return null;

  const pendingRequests = requests.filter((r) => r.status === 'pending');
  const otherRequests = requests.filter((r) => r.status !== 'pending');
  const displayedRequests = activeTab === 'pending' ? pendingRequests : requests;

  const getCostFor = (req: CustomRewardRequest) => {
    if (editedCosts[req.id] !== undefined) {
      return editedCosts[req.id];
    }
    return req.proposedCost;
  };

  const handleCostChange = (reqId: string, delta: number) => {
    playPopSound();
    setEditedCosts((prev) => {
      const current = prev[reqId] !== undefined ? prev[reqId] : requests.find((r) => r.id === reqId)?.proposedCost || 20;
      const next = Math.max(1, Math.min(200, current + delta));
      return { ...prev, [reqId]: next };
    });
  };

  const handleApprove = (req: CustomRewardRequest) => {
    playFanfareSound();
    const finalCost = getCostFor(req);
    onApproveRequest(req.id, finalCost);
  };

  const handleReject = (reqId: string) => {
    playPopSound();
    onRejectRequest(reqId);
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-3 md:p-4 bg-black/60 backdrop-blur-xs select-none overflow-y-auto" dir="rtl">
      <div className="relative w-full max-w-lg rounded-3xl p-5 md:p-6 bg-[#FAF5EF] border-4 border-black comic-shadow my-auto text-[#2B1B15]">
        
        {/* Close Button */}
        <button
          onClick={() => {
            playPopSound();
            onClose();
          }}
          type="button"
          className="absolute top-3 left-3 w-8 h-8 rounded-full bg-stone-200 hover:bg-stone-300 text-stone-700 font-black flex items-center justify-center border-2 border-black cursor-pointer text-sm"
        >
          ✕
        </button>

        {/* Header */}
        <div className="text-center mb-4">
          <div className="inline-flex items-center gap-1.5 bg-blue-100 text-blue-900 border border-blue-400 rounded-full px-3 py-0.5 text-xs font-black mb-1.5">
            <span>👨‍👩‍👧</span>
            <span>بوابة ولي الأمر</span>
          </div>
          <h3 className="text-2xl font-black text-[#231510]">
            مراجعة طلبات الهدايا الخاصة
          </h3>
          <p className="text-xs text-[#5C3224] mt-0.5">
            يمكنك الموافقة على الهدايا المقترحة من طفلك وتعديل عدد النجوم المطلوبة لاستبدالها.
          </p>
        </div>

        {/* Tab Switcher */}
        <div className="flex gap-2 mb-3 bg-stone-200/80 p-1 rounded-2xl border border-black/20">
          <button
            type="button"
            onClick={() => {
              playPopSound();
              setActiveTab('pending');
            }}
            className={`flex-1 py-1.5 rounded-xl text-xs font-black transition-all cursor-pointer flex items-center justify-center gap-1.5 ${
              activeTab === 'pending'
                ? 'bg-white text-[#2B1B15] shadow-xs border border-black/20'
                : 'text-stone-600 hover:text-stone-900'
            }`}
          >
            <span>طلبات جديدة بانتظارك</span>
            {pendingRequests.length > 0 && (
              <span className="w-5 h-5 bg-red-500 text-white text-[10px] rounded-full flex items-center justify-center font-bold">
                {pendingRequests.length}
              </span>
            )}
          </button>

          <button
            type="button"
            onClick={() => {
              playPopSound();
              setActiveTab('all');
            }}
            className={`flex-1 py-1.5 rounded-xl text-xs font-black transition-all cursor-pointer ${
              activeTab === 'all'
                ? 'bg-white text-[#2B1B15] shadow-xs border border-black/20'
                : 'text-stone-600 hover:text-stone-900'
            }`}
          >
            جميع الطلبات ({requests.length})
